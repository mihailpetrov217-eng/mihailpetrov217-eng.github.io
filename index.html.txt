        let displayedTotalUsd = 0;
        let balanceAnimFrame = null;

        function animateNetWorthTo(targetUsd, instant = false) {
            const el = document.getElementById('netWorthValue');
            if (!el) return;

            if (state.privacyMode) {
                displayedTotalUsd = targetUsd;
                el.textContent = '••••••••';
                return;
            }

            if (instant || Math.abs(targetUsd - displayedTotalUsd) < 0.005) {
                displayedTotalUsd = targetUsd;
                el.textContent = formatDisplayCurrency(targetUsd);
                return;
            }

            if (balanceAnimFrame) cancelAnimationFrame(balanceAnimFrame);
            const startUsd = displayedTotalUsd;
            const diff = targetUsd - startUsd;
            const duration = 520;
            const startTime = performance.now();

            function step(now) {
                const elapsed = now - startTime;
                const progress = Math.min(1, elapsed / duration);
                // Выход по плавной кривой easeOutQuart
                const eased = 1 - Math.pow(1 - progress, 4);
                displayedTotalUsd = startUsd + diff * eased;
                el.textContent = formatDisplayCurrency(displayedTotalUsd);

                if (progress < 1) {
                    balanceAnimFrame = requestAnimationFrame(step);
                } else {
                    displayedTotalUsd = targetUsd;
                    el.textContent = formatDisplayCurrency(targetUsd);
                }
            }

            balanceAnimFrame = requestAnimationFrame(step);
        }

        async function fetchLiveRates(manual = false) {
            const icon = document.getElementById('syncIcon');
            if (icon) icon.classList.add('animate-spin');

            try {
                const res = await fetch('https://api.coingecko.com/api/v3/simple/price?ids=bitcoin,ethereum,tether,toncoin,solana,tron,binancecoin&vs_currencies=usd,rub,eur&include_24hr_change=true', { cache: 'no-store' });
                if (!res.ok) throw new Error('Network error');
                const data = await res.json();

                if (data.bitcoin?.usd && data.bitcoin?.rub) {
                    state.fxRates.RUB = data.bitcoin.rub / data.bitcoin.usd;
                    state.fxRates.EUR = data.bitcoin.eur / data.bitcoin.usd;
                    state.prices.RUB.usd = 1 / state.fxRates.RUB;
                    state.prices.EUR.usd = 1 / state.fxRates.EUR;
                }

                Object.keys(ASSETS).forEach(sym => {
                    const gId = ASSETS[sym].geckoId;
                    if (gId && data[gId]) {
                        state.prices[sym] = {
                            usd: data[gId].usd,
                            change24h: data[gId].usd_24h_change || 0
                        };
                    }
                });

                saveStorage();
                renderApp();
                if (manual) showToast('Биржевые котировки обновлены');
            } catch (e) {
                if (manual) showToast('Используются локальные котировки');
            } finally {
                if (icon) setTimeout(() => icon.classList.remove('animate-spin'), 400);
            }
        }

        function formatDisplayCurrency(usdValue) {
            if (state.privacyMode) return '••••••••';
            const cur = state.displayCurrency;
            const rate = state.fxRates[cur] || 1;
            const converted = usdValue * rate;

            if (cur === 'RUB') {
                return new Intl.NumberFormat('ru-RU', { style: 'currency', currency: 'RUB', maximumFractionDigits: 0 }).format(converted);
            }
            if (cur === 'EUR') {
                return new Intl.NumberFormat('de-DE', { style: 'currency', currency: 'EUR', minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(converted);
            }
            return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(converted);
        }

        function formatUnitPrice(usdPrice) {
            if (state.displayCurrency === 'RUB') {
                const r = usdPrice * state.fxRates.RUB;
                return r >= 10 ? `₽${r.toLocaleString('ru-RU', { maximumFractionDigits: 2 })}` : `₽${r.toFixed(4)}`;
            }
            return usdPrice >= 1 ? `$${usdPrice.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}` : `$${usdPrice.toFixed(4)}`;
        }

        function formatQty(qty, symbol) {
            if (state.privacyMode) return '•••';
            const dec = ASSETS[symbol]?.decimals ?? 2;
            return Number(qty).toLocaleString('en-US', { minimumFractionDigits: 0, maximumFractionDigits: dec });
        }

        function computePortfolioTotals() {
            let totalUsd = 0;
            let pnl24hUsd = 0;

            Object.keys(ASSETS).forEach(sym => {
                const qty = parseFloat(state.holdings[sym] || 0);
                const price = state.prices[sym]?.usd || 0;
                const chg = state.prices[sym]?.change24h || 0;
                const val = qty * price;
                totalUsd += val;
                pnl24hUsd += val * (chg / 100);
            });

            return { totalUsd, pnl24hUsd };
        }

            const { totalUsd, pnl24hUsd } = computePortfolioTotals();
            animateNetWorthTo(totalUsd, instant);

                pnlEl.textContent = `${formatDisplayCurrency(0)} (0.00%)`;
                pnlEl.className = 'text-[13px] font-medium tabular-nums text-[#8E8E93]';
                return;
            }

            const pct = (pnl24hUsd / totalUsd) * 100;
            const sign = pnl24hUsd >= 0 ? '+' : '-';
            pnlEl.textContent = `${sign}${formatDisplayCurrency(Math.abs(pnl24hUsd))} (${pnl24hUsd >= 0 ? '+' : ''}${pct.toFixed(2)}%)`;
            pnlEl.className = `text-[13px] font-medium tabular-nums ${pnl24hUsd >= 0 ? 'text-[#30D158]' : 'text-[#FF453A]'}`;
        }

        function drawChartSvg(svgId, seedString, isPositive = true, isFlatZero = false) {
            const svg = document.getElementById(svgId);
            if (!svg) return;

            if (isFlatZero) {
                svg.innerHTML = `
                    <line x1="0" y1="74" x2="360" y2="74" stroke="#2C2C2E" stroke-width="2" stroke-dasharray="4 4" />
                    <line x1="0" y1="74" x2="360" y2="74" stroke="#636366" stroke-width="2" class="chart-line-anim" />
                `;
                return;
            }

            const color = isPositive ? '#30D158' : '#FF453A';
            const pointsCount = 34;
            let hash = 0;
            for (let i = 0; i < seedString.length; i++) {
                hash = ((hash << 5) - hash) + seedString.charCodeAt(i);
                hash |= 0;
            }

            const coords = [];
            let yVal = isPositive ? 68 : 28;
            for (let i = 0; i < pointsCount; i++) {
                const step = ((Math.sin(i * 1.4 + hash) + Math.cos(i * 0.7 - hash)) * 5.2) + (isPositive ? -0.9 : 0.9);
                yVal = Math.max(10, Math.min(82, yVal + step));
                const x = (i / (pointsCount - 1)) * 360;
                coords.push(`${x.toFixed(1)},${yVal.toFixed(1)}`);
            }

            const gradId = 'grad_' + svgId;
            svg.innerHTML = `
                <defs>
                    <linearGradient id="${gradId}" x1="0" y1="0" x2="0" y2="1">
                        <stop offset="0%" stop-color="${color}" stop-opacity="0.24"/>
                        <stop offset="100%" stop-color="${color}" stop-opacity="0.0"/>
                    </linearGradient>
                </defs>
                <polygon class="chart-fill-anim" points="0,95 ${coords.join(' ')} 360,95" fill="url(#${gradId})"/>
                <polyline class="chart-line-anim" fill="none" stroke="${color}" stroke-width="2.2" stroke-linecap="round" stroke-linejoin="round" points="${coords.join(' ')}"/>
            `;
        }

        function executeGrantAsset() {
            const sym = document.getElementById('grantAssetSelect').value;
            const mode = document.getElementById('grantInputCurrency').value;
            const rawAmt = parseFloat(document.getElementById('grantAmountInput').value);

            if (isNaN(rawAmt) || rawAmt <= 0) {
                showToast('Введите корректную сумму', true);
                return;
            }

            const assetUsdPrice = state.prices[sym]?.usd || 1;
            let tokensToAdd = 0;
            let usdEquivalent = 0;

            if (mode === 'USD') {
                usdEquivalent = rawAmt;
                tokensToAdd = rawAmt / assetUsdPrice;
            } else if (mode === 'RUB') {
                usdEquivalent = rawAmt / state.fxRates.RUB;
                tokensToAdd = usdEquivalent / assetUsdPrice;
            } else {
                tokensToAdd = rawAmt;
                usdEquivalent = rawAmt * assetUsdPrice;
            }

            const dec = ASSETS[sym].decimals;
            state.holdings[sym] = parseFloat(((state.holdings[sym] || 0) + tokensToAdd).toFixed(dec));

            const nowStr = new Date().toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' });
            state.history.unshift({
                id: 'tx_' + Date.now(),
                type: 'in',
                symbol: sym,
                qty: tokensToAdd,
                usdVal: usdEquivalent,
                meta: ASSETS[sym].category === 'stocks' ? `Брокерское зачисление ${sym}` : `Пополнение ${sym}`,
                time: `Сегодня, ${nowStr}`
            });

            document.getElementById('grantAmountInput').value = '';
            saveStorage();
            renderApp(false);
            closeModal('adminModal');
            showToast(`Зачислено +${formatQty(tokensToAdd, sym)} ${sym}`);
        }

        function recalcSwap() {
            const fromSym = document.getElementById('swapFromSelect').value || 'USDT';
            const toSym = document.getElementById('swapToSelect').value || 'BTC';
            const qty = parseFloat(document.getElementById('swapFromInput').value) || 0;

            document.getElementById('swapAvailSpan').textContent = `${formatQty(state.holdings[fromSym] || 0, fromSym)} ${fromSym}`;

            const pFrom = state.prices[fromSym]?.usd || 1;
            const pTo = state.prices[toSym]?.usd || 1;
            const cross = pFrom / pTo;

            document.getElementById('swapCrossRateText').textContent = `Рыночный курс: 1 ${fromSym} = ${cross.toFixed(6)} ${toSym}`;
            document.getElementById('swapToOutput').value = qty > 0 ? (qty * cross).toFixed(ASSETS[toSym].decimals) : '';
        }

        function executeSwapOrder() {
            const fromSym = document.getElementById('swapFromSelect').value;
            const toSym = document.getElementById('swapToSelect').value;
            const qty = parseFloat(document.getElementById('swapFromInput').value);

            if (fromSym === toSym) {
                showToast('Выберите разные активы', true);
                return;
            }
            if (isNaN(qty) || qty <= 0 || qty > (state.holdings[fromSym] || 0)) {
                showToast('Недостаточно средств для ордера', true);
                return;
            }

            const pFrom = state.prices[fromSym]?.usd || 1;
            const pTo = state.prices[toSym]?.usd || 1;
            const outQty = qty * (pFrom / pTo);

            state.holdings[fromSym] = parseFloat((state.holdings[fromSym] - qty).toFixed(ASSETS[fromSym].decimals));
            state.holdings[toSym] = parseFloat(((state.holdings[toSym] || 0) + outQty).toFixed(ASSETS[toSym].decimals));

            const nowStr = new Date().toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' });
            state.history.unshift({
                id: 'tx_' + Date.now(),
                type: 'swap',
                symbol: toSym,
                qty: outQty,
                usdVal: qty * pFrom,
                meta: `Обмен ${fromSym} → ${toSym}`,
                time: `Сегодня, ${nowStr}`
            });

            document.getElementById('swapFromInput').value = '';
            saveStorage();
            renderApp(false);
            closeModal('swapModal');
            showToast(`Куплено +${formatQty(outQty, toSym)} ${toSym}`);
        }

        // QR и реквизиты
        function updateReceiveQr() {
            const sym = document.getElementById('receiveAssetSelect').value || 'USDT';
            const addr = ASSETS[sym].address;
            document.getElementById('receiveAddrString').textContent = addr;

            const svg = document.getElementById('receiveQrSvg');
            let html = '<rect width="33" height="33" fill="white"/>';
            const finder = (x, y) => {
                html += `<rect x="${x}" y="${y}" width="7" height="7" fill="black"/><rect x="${x+1}" y="${y+1}" width="5" height="5" fill="white"/><rect x="${x+2}" y="${y+2}" width="3" height="3" fill="black"/>`;
            };
            finder(2, 2); finder(24, 2); finder(2, 24);
            let h = 0;
            for (let i = 0; i < addr.length; i++) h = ((h << 5) - h) + addr.charCodeAt(i);
            for (let y = 2; y < 31; y++) {
                for (let x = 2; x < 31; x++) {
                    if ((x < 10 && y < 10) || (x > 22 && y < 10) || (x < 10 && y > 22)) continue;
                    if (Math.abs((x * 19 + y * 37 + h) % 5) < 2) {
                        html += `<rect x="${x}" y="${y}" width="1" height="1" fill="black"/>`;
                    }
                }
            }
            svg.innerHTML = html;
        }
